// Fungsi untuk menormalkan berbagai format tanggal dari Sheet ke YYYY-MM-DD function ubahFormatUntukPerbandingan(tglStr) { if (!tglStr) return ""; let s = tglStr.toString().trim(); // Jika format dari Sheet mengandung garis miring (DD/MM/YYYY atau MM/DD/YYYY) if (s.includes("/")) { let parts = s.split("/"); if (parts.length === 3) { // Cek apakah formatnya DD/MM/YYYY (misal: 28/09/2026) let p1 = parts[0].padStart(2, '0'); let p2 = parts[1].padStart(2, '0'); let p3 = parts[2]; // Jika tahun ada di posisi ke-3 (YYYY) if (p3.length === 4) { return `${p3}-${p2}-${p1}`; // Menjadi YYYY-MM-DD } // Jika tahun ada di posisi pertama (YYYY/MM/DD) if (p1.length === 4) { return `${p1}-${p2}-${p3}`; } } } // Jika format sudah YYYY-MM-DD if (s.includes("-")) { let parts = s.split("-"); if (parts.length === 3 && parts[0].length === 4) { return s; } } return s; } function filterData() { let supplierFilter = document.getElementById("pilihSupplier").value; let tanggalFilter = document.getElementById("pilihTanggal").value; // Format: YYYY-MM-DD dari input browser if (!tanggalFilter) { document.getElementById("isiTabel").innerHTML = `Silakan pilih tanggal kedatangan terlebih dahulu.`; document.getElementById("statsSummary").innerText = ""; return; } let dataFiltered = semuaDataGlobal.filter(row => { let matchSupplier = true; let matchTanggal = false; if (supplierFilter !== "semua") { let supData = (row.supplier || "").trim().toUpperCase(); let supFilter = supplierFilter.trim().toUpperCase(); matchSupplier = supData.includes(supFilter) || supFilter.includes(supData); } // Normalisasi tanggal dari baris data agar formatnya sama (YYYY-MM-DD) let tglBarisNormalized = ubahFormatUntukPerbandingan(row.tanggal); // Cocokkan string tanggal yang dinormalkan dengan tanggal input browser if (tglBarisNormalized === tanggalFilter) { matchTanggal = true; } return matchSupplier && matchTanggal; }); tampilkanData(dataFiltered); }
